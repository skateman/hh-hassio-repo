#include <tunables/global>

  #include <abstractions/base>

  # Capabilities
  file,
  signal (send) set=(kill,term,int,hup,cont),
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Access to options.json and other files within your addon
  /data/** rw,

  # Start new profile for service
  /usr/bin/python3* cx -> mcp-orchestrator,

  profile mcp-orchestrator flags=(attach_disconnected,mediate_deleted,complain) {
    #include <abstractions/base>

    # Receive signals from S6-Overlay
    signal (receive) peer=*_example,

    # Network access (Azure OpenAI, MCP servers, Supervisor API, inbound on 11434)
    network inet stream,
    network inet dgram,
    network inet6 stream,
    network inet6 dgram,

    # Access to options.json and other files within your addon
    /data/** rw,

    # Python interpreter and standard library
    /usr/bin/python3* rix,
    /usr/lib/python3*/** r,
    /usr/lib/python3*/lib-dynload/** mr,

    # Installed pip packages
    /usr/lib/python3*/site-packages/** r,

    # Application code
    /app/** r,

    # Python runtime needs
    /etc/hosts r,
    /etc/resolv.conf r,
    /etc/ssl/** r,
    /usr/share/ca-certificates/** r,
    /etc/ssl/certs/ca-certificates.crt r,
    /proc/sys/net/core/somaxconn r,
    /tmp/** rwk,
    /dev/null rw,
    /dev/urandom r,
  }
}
